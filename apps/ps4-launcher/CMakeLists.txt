cmake_minimum_required(VERSION 3.16)
project(odium_ps4_transfer_core VERSION 0.1.0 LANGUAGES CXX)
# HOST test target, not a PS4 PKG compiler.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(odium_transfer STATIC src/transfer_plan.cpp src/download_journal.cpp)
target_include_directories(odium_transfer PUBLIC include)
if(MSVC)
  target_compile_options(odium_transfer PRIVATE /W4)
else()
  target_compile_options(odium_transfer PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(test_transfer_plan tests/test_transfer_plan.cpp)
target_link_libraries(test_transfer_plan PRIVATE odium_transfer)
add_test(NAME transfer_plan COMMAND test_transfer_plan)
add_executable(test_download_journal tests/test_download_journal.cpp)
target_link_libraries(test_download_journal PRIVATE odium_transfer)
add_test(NAME download_journal COMMAND test_download_journal)
